        function checkCode() {
            const codeInput = document.getElementById('criminal-code').value.toLowerCase().trim();
            const errorMsg = document.getElementById('error-msg');
            const loginScreen = document.getElementById('login-screen');
            const mainContent = document.getElementById('main-content');
            const welcomeText = document.getElementById('welcome-text');

            if (members[codeInput]) {
                const user = members[codeInput];
                // إخفاء شاشة الدخول وإظهار المحتوى
                loginScreen.style.height = "40vh";
                document.querySelector('.login-box').style.display = "none";
                errorMsg.style.display = "none";

                welcomeText.innerHTML = `⚠️ مرحباً بالعضو: <b>${user.name}</b><br>رتبتك الحالية: <span style="color:var(--gold)">${user.rank}</span>`;

                errorMsg.style.display = "block";
                document.getElementById('criminal-code').style.borderColor = "red";
