    <script>
        const users = {
            "ss-0-112": { name: "اللاندر", rank: "رئيس المنظمة" },
            "ss-01-27": { name: "حمد بن مترك", rank: "عضو مافيا" },
            "ss-11-29": { name: "لافنير", rank: "ابن عم اللاندر" },
            "ss-00-28": { name: "طراد بن مسبق", rank: "عضو مافيا" },
            "ss-41-30": { name: "عبدالجليل ششه", rank: "عضو مافيا" },
            "ss-92-31": { name: "ابو شهاب", rank: "عضو مافيا" },
            "ll-12-65": { name: "عبدالله عبدلي", rank: "عضو مافيا" }
        };

        function login() {
            const code = document.getElementById('code-input').value.toLowerCase();
            if (users[code]) {
                document.getElementById('login-screen').style.display = 'none';
                document.getElementById('main-content').style.display = 'block';
                document.getElementById('user-banner').innerHTML = `أهلاً بك يا <b>${users[code].name}</b><br>رتبتك الحالية: ${users[code].rank}`;
            } else {
                document.getElementById('error').style.display = 'block';
            }
        }

        function showPage(pageId) {
            document.querySelectorAll('.page').forEach(p => p.style.display = 'none');
            document.querySelectorAll('nav button').forEach(b => b.classList.remove('active'));
            document.getElementById(pageId).style.display = 'block';
            event.target.classList.add('active');
        }
    </script>
